Operadores aritméticos

```javascript

	console.log(5 + 2);
	
	console.log(5 - 2);
	
	console.log(5 * 2);
	
	console.log(5 / 2);
	
	console.log(5 % 2);
	
	  
	
	let n1 = 9;
	
	let n2 = 2;
	
	  
	
	console.log(n1 + n2);
	
	console.log(n1 - n2);
	
	console.log(n1 * n2);
	
	console.log(n1 / n2);
	
	  
	
	console.log((n1 % n2) + (n2 * n1) / n2);
```

```javascript

	console.log(1 < 2);
	
	console.log(1 >= 1);
	
	console.log(1 <= 1);

	console.log(1 != 1);
```

Operadores relacionais #2

```javascript

	let n1 = 123;
	
	let n2 = 123;
	
	let s1 = "123";
	
	  
	
	console.log("Igualdade.....");
	
	console.log(n1 == s1);
	
	console.log(n1 === n2);
	
	console.log(n1 === s1);
	
	  
	
	console.log("Diferença.....");
	
	console.log(n1 != s1);
	
	console.log(n1 !== s1);
```

Operadores lógicos 

```javascript

	// Tabela Verdade AND (&&)

	console.log("Tabela Verdade AND (&&)....");
	
	console.log(true && true); // true
	
	console.log(true && false); // false
	
	console.log(false && true); // false
	
	console.log(false && false); // false
	
	  
	
	// Tabela Verdade OR (||)
	
	console.log("Tabela Verdade OR (||)....");
	
	console.log(true || true); // true
	
	console.log(true || false); // true
	
	console.log(false || true); // true
	
	console.log(false || false); // false
	
	  
	
	// Tabela Verdade XOR (^)
	
	console.log("Tabela Verdade XOR (^)....");
	
	console.log(true != true); // false
	
	console.log(false != true); // true
	
	console.log(true != false); // true
	
	console.log(false != false); // false
	
	  
	
	// Tabela Verdade NOT (!)
	
	console.log("Tabela Verdade NOT (!)....");
	
	console.log(!true); // false
	
	console.log(!false); // true
```
